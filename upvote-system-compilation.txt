// ============================================================
// PROJECT: TANYA USTADZ V3
// FILE: composables/useUpvote.ts
// DESC: Manages upvote state per question.
//       - Optimistic UI (immediate count change, rollback on error)
//       - Voted status always synced atomically: hasVoted + localStorage
//         updated together via syncHasVoted() — icon can never be "left behind"
//       - Handles 'force_retracted' from the 23505 fallback DELETE path
//       - Supabase Realtime: listens for UPDATE on questions row so
//         upvote_count stays live when other users vote
//
// STATE SYNC CONTRACT:
//   Every terminal branch of toggleUpvote() must call syncHasVoted()
//   exactly once. This writes hasVoted.value AND localStorage atomically,
//   so they are always consistent regardless of which code path was taken.
//
// ACTION MAP (from vote.post.ts):
//   'voted'           → INSERT ok        → hasVoted=true,  count stays +1
//   'retracted'       → DELETE ok        → hasVoted=false, count stays -1
//   'force_retracted' → 23505 → DELETE   → hasVoted=false, count rollback to prev-1
//                       (server found a stale row we didn't know about)
//
// DB CONTRACT (question_votes):
//   - UNIQUE (question_id, fingerprint)
//   - TRIGGER trg_sync_upvotes AFTER INSERT OR DELETE
//     → sync_upvote_count() → updates questions.upvote_count
//     → we NEVER manually touch upvote_count here
// ============================================================

import type { RealtimeChannel } from '@supabase/supabase-js'

const VOTED_KEY = 'tanya_ustadz_voted' // localStorage → JSON array of question IDs

// ── localStorage helpers ──────────────────────────────────────

function loadVotedSet(): Set<string> {
  if (typeof window === 'undefined') return new Set()
  try {
    const raw = localStorage.getItem(VOTED_KEY)
    return raw ? new Set(JSON.parse(raw) as string[]) : new Set()
  } catch {
    return new Set()
  }
}

function saveVotedSet(set: Set<string>): void {
  if (typeof window === 'undefined') return
  localStorage.setItem(VOTED_KEY, JSON.stringify([...set]))
}

// ── Composable ────────────────────────────────────────────────

export function useUpvote(questionId: string, initialCount: number) {
  // Auto-imported by Nuxt — no explicit import needed
  const { getFingerprint }  = useFingerprint()
  const supabase            = useSupabaseClient()

  const count             = ref(initialCount)
  const hasVoted          = ref(false)
  const isPending         = ref(false)
  const isRealtimeSyncing = ref(false)

  let channel: RealtimeChannel | null = null

  // ── Atomic state sync helper ──────────────────────────────
  // ALWAYS use this instead of setting hasVoted.value directly.
  // Keeps reactive state and localStorage in lockstep.
  function syncHasVoted(value: boolean): void {
    hasVoted.value = value
    const voted = loadVotedSet()
    value ? voted.add(questionId) : voted.delete(questionId)
    saveVotedSet(voted)
  }

  // ── Hydrate voted state from localStorage on mount ────────
  onMounted(() => {
    hasVoted.value = loadVotedSet().has(questionId)
    subscribeRealtime()
  })

  onUnmounted(() => {
    if (channel) {
      supabase.removeChannel(channel)
      channel = null
    }
  })

  // ── Realtime subscription ─────────────────────────────────
  // Listens to UPDATE on questions (not question_votes) because
  // trg_sync_upvotes writes the final upvote_count to questions.
  // One channel per question — safe on list pages with many buttons.
  function subscribeRealtime() {
    channel = supabase
      .channel(`upvote:${questionId}`)
      .on(
        'postgres_changes',
        {
          event:  'UPDATE',
          schema: 'public',
          table:  'questions',
          filter: `id=eq.${questionId}`,
        },
        (payload) => {
          const newCount: number | undefined = payload.new?.upvote_count

          // Guard 1: malformed payload
          if (typeof newCount !== 'number') return

          // Guard 2: own request in-flight — optimistic value is already correct;
          // accepting this push would double-apply the change
          if (isPending.value) return

          // Guard 3: nothing changed (Supabase fires UPDATE for any column change)
          if (newCount === count.value) return

          isRealtimeSyncing.value = true
          count.value = newCount
          setTimeout(() => { isRealtimeSyncing.value = false }, 800)
        }
      )
      .subscribe()
  }

  // ── Toggle ────────────────────────────────────────────────
  async function toggleUpvote() {
    if (isPending.value) return

    const fp = await getFingerprint()
    if (!fp) return

    const wasVoted  = hasVoted.value
    const prevCount = count.value

    // Optimistic update — mirrors the DB trigger:
    //   INSERT → trigger increments upvote_count → +1
    //   DELETE → trigger decrements upvote_count → -1
    hasVoted.value  = !wasVoted
    count.value     = wasVoted ? prevCount - 1 : prevCount + 1
    isPending.value = true

    try {
      const data = await $fetch<{ success: boolean; action: string }>(
        '/api/questions/vote',
        {
          method: 'POST',
          body: { question_id: questionId, fingerprint: fp },
        }
      )

      if (data.action === 'voted') {
        // INSERT ok → trigger fired → +1 confirmed
        // Optimistic +1 is correct
        syncHasVoted(true)

      } else if (data.action === 'retracted') {
        // DELETE ok → trigger fired → -1 confirmed
        // Optimistic -1 is correct
        syncHasVoted(false)

      } else if (data.action === 'force_retracted') {
        // 23505 race: server found a stale existing row and deleted it.
        // We thought the user wasn't voted (wasVoted=false), but they were.
        // Correct the count: undo our optimistic +1, then apply the -1 from delete.
        // Net result: prevCount - 1
        syncHasVoted(false)
        count.value = prevCount - 1
      }
      // No else branch needed — vote.post.ts throws on all other error conditions

    } catch {
      // Network / server error — full rollback, no DB change occurred
      hasVoted.value = wasVoted   // direct write on rollback — no localStorage change needed
      count.value    = prevCount
    } finally {
      isPending.value = false
    }
  }

  return { count, hasVoted, isPending, isRealtimeSyncing, toggleUpvote }
}

<script setup lang="ts">

const props = withDefaults(
  defineProps<{
    questionId: string
    initialCount: number
    size?: 'sm' | 'md'
  }>(),
  { size: 'sm' }
)

const { count, hasVoted, isPending, isRealtimeSyncing, toggleUpvote } = useUpvote(
  props.questionId,
  props.initialCount
)

const isMd = computed(() => props.size === 'md')

// ── Hover ─────────────────────────────────────────────────────
const isHovered = ref(false)

// ── Animation state ───────────────────────────────────────────
const animState = ref<'idle' | 'voting' | 'unvoting'>('idle')

// Burst particles for unvote
const particles = ref<{ id: number; angle: number }[]>([])
let particleId = 0

function spawnParticles() {
  const n = 6
  particles.value = Array.from({ length: n }, (_, i) => ({
    id: particleId++,
    angle: (360 / n) * i,
  }))
  setTimeout(() => { particles.value = [] }, 700)
}

// ── Click ─────────────────────────────────────────────────────
async function handleClick(e: Event) {
  e.preventDefault()
  e.stopPropagation()
  if (isPending.value) return

  if (hasVoted.value) {
    animState.value = 'unvoting'
    spawnParticles()
    setTimeout(() => { animState.value = 'idle' }, 500)
  } else {
    animState.value = 'voting'
    setTimeout(() => { animState.value = 'idle' }, 400)
  }

  await toggleUpvote()
}
</script>

// ============================================================
// PROJECT: TANYA USTADZ V3
// FILE: server/api/questions/vote.post.ts
// DESC: Submit or retract an upvote for a question.
//
// DB CONTRACT (question_votes):
//   - PRIMARY KEY: id (uuid, gen_random_uuid())
//   - UNIQUE:      (question_id, fingerprint)
//   - FK:          question_id → questions(id) ON DELETE CASCADE
//   - INDEX:       idx_votes_fingerprint ON (question_id, fingerprint)
//   - TRIGGER:     trg_sync_upvotes AFTER INSERT OR DELETE
//                  → sync_upvote_count() → updates questions.upvote_count
//                  → we NEVER manually touch upvote_count here
//
// VOTE FLOW:
//   1. Validate inputs
//   2. Confirm question exists and is not rejected
//   3. SELECT to check vote existence (.maybeSingle())
//      — avoids relying on deletedCount which RLS can silently block
//   4a. Row EXISTS  → DELETE  → action: 'retracted'
//   4b. Row MISSING → INSERT
//       - Success          → action: 'voted'
//       - 23505 (race)     → another request inserted between our
//                            SELECT and INSERT; fall back to DELETE
//                            → action: 'force_retracted'
//       - Other error      → 500
// ============================================================

import { defineEventHandler, readBody, createError } from "h3";
import { serverSupabaseClient } from "#supabase/server";

export default defineEventHandler(async (event) => {
  const body = await readBody(event);

  const questionId: string  = (body?.question_id ?? "").trim();
  const fingerprint: string = (body?.fingerprint  ?? "").trim();

  // ── Validation ────────────────────────────────────────────
  if (!questionId) {
    throw createError({ statusCode: 400, message: "question_id wajib diisi." });
  }
  if (!fingerprint || fingerprint === "ssr-placeholder") {
    throw createError({ statusCode: 400, message: "Fingerprint tidak valid." });
  }

  const supabase = await serverSupabaseClient(event);

  // ── Confirm question exists and is voteable ───────────────
  const { data: question, error: fetchError } = await supabase
    .from("questions")
    .select("id, status")
    .eq("id", questionId)
    .single();

  if (fetchError || !question) {
    throw createError({ statusCode: 404, message: "Pertanyaan tidak ditemukan." });
  }
  if (question.status === "rejected") {
    throw createError({ statusCode: 403, message: "Pertanyaan ini tidak dapat di-upvote." });
  }

  // ── Step 3: Check vote existence via SELECT ───────────────
  // We use .maybeSingle() instead of relying on deletedCount from
  // .delete({ count: 'exact' }), because RLS policies can silently
  // block the DELETE and return count=0 even when a row exists,
  // causing the server to incorrectly attempt an INSERT.
  const { data: existingVote, error: selectError } = await supabase
    .from("question_votes")
    .select("id")
    .eq("question_id", questionId)
    .eq("fingerprint", fingerprint)
    .maybeSingle();

  if (selectError) {
    throw createError({ statusCode: 500, message: "Gagal memeriksa status vote." });
  }

  // ── Step 4a: Row exists → DELETE (unvote) ────────────────
  if (existingVote) {
    const { error: deleteError } = await supabase
      .from("question_votes")
      .delete()
      .eq("question_id", questionId)
      .eq("fingerprint", fingerprint);

    if (deleteError) {
      throw createError({ statusCode: 500, message: "Gagal menghapus vote." });
    }

    // trg_sync_upvotes fired → upvote_count decremented by trigger
    return { success: true, action: "retracted" };
  }

  // ── Step 4b: Row missing → INSERT (vote) ─────────────────
  const { error: insertError } = await supabase
    .from("question_votes")
    .insert({ question_id: questionId, fingerprint });

  if (!insertError) {
    // trg_sync_upvotes fired → upvote_count incremented by trigger
    return { success: true, action: "voted" };
  }

  // ── Step 4b fallback: 23505 unique violation ──────────────
  // Another concurrent request inserted the row between our SELECT
  // and this INSERT. The user has effectively already voted, so we
  // treat this as an unvote intent and delete the row they own.
  if (insertError.code === "23505") {
    const { error: fallbackDeleteError } = await supabase
      .from("question_votes")
      .delete()
      .eq("question_id", questionId)
      .eq("fingerprint", fingerprint);

    if (fallbackDeleteError) {
      // DELETE also failed — surface a clean error rather than hanging
      throw createError({ statusCode: 500, message: "Gagal memproses vote (konflik data)." });
    }

    // trg_sync_upvotes fired on the fallback DELETE
    return { success: true, action: "force_retracted" };
  }

  throw createError({ statusCode: 500, message: "Gagal menyimpan vote." });
});
